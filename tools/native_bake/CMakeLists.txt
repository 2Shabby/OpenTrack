cmake_minimum_required(VERSION 3.20)
project(opentrack_native_bake LANGUAGES CXX)
set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
set(GODOTCPP_API_VERSION 4.7 CACHE STRING "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/profile.json" CACHE PATH "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../extensions/godot-cpp" godot-cpp)
add_library(native_bake SHARED native_bake.cpp)
target_link_libraries(native_bake PRIVATE godot-cpp)
target_compile_features(native_bake PRIVATE cxx_std_17)
target_compile_options(native_bake PRIVATE -O3 -ffp-contract=off)
set_target_properties(native_bake PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../.stage_authoring/native")
